import React, { useEffect, useRef, useState } from "react";
import {
  View,
  Text,
  TouchableOpacity,
  StyleSheet,
  Animated,
  Dimensions,
} from "react-native";
import { Camera } from "expo-camera";
import * as ImageManipulator from "expo-image-manipulator";
import { NavigationContainer } from "@react-navigation/native";
import { createStackNavigator } from "@react-navigation/stack";
import Slider from "@react-native-community/slider";

const Stack = createStackNavigator();
const { width } = Dimensions.get("window");

function CameraScreen({ navigation }) {
  const cameraRef = useRef(null);
  const [hasPermission, setHasPermission] = useState(null);
  const [zoom, setZoom] = useState(0);
  const [displayZoom, setDisplayZoom] = useState(1);
  const blinkAnim = useRef(new Animated.Value(1)).current;

  // Permisos
  useEffect(() => {
    (async () => {
      const { status } = await Camera.requestCameraPermissionsAsync();
      setHasPermission(status === "granted");
    })();
  }, []);

  // Animación Blink AI
  useEffect(() => {
    Animated.loop(
      Animated.sequence([
        Animated.timing(blinkAnim, {
          toValue: 0.2,
          duration: 800,
          useNativeDriver: true,
        }),
        Animated.timing(blinkAnim, {
          toValue: 1,
          duration: 800,
          useNativeDriver: true,
        }),
      ])
    ).start();
  }, []);

  // Zoom inteligente (0 a 120x)
  const handleZoom = (value) => {
    let mapped = value * 0.99; // límite cámara
    setZoom(mapped);

    // Simulación 0.6x a 120x
    let zoomLevel = 0.6 + value * 119.4;
    setDisplayZoom(zoomLevel.toFixed(1));
  };

  // Captura con “IA simulada”
  const takePhoto = async () => {
    if (cameraRef.current) {
      let photo = await cameraRef.current.takePictureAsync();

      // Post-procesado IA (nitidez simulada)
      let enhanced = await ImageManipulator.manipulateAsync(
        photo.uri,
        [],
        {
          compress: 1,
          format: ImageManipulator.SaveFormat.JPEG,
        }
      );

      console.log("Foto procesada IA:", enhanced.uri);
    }
  };

  if (hasPermission === null) return <View />;
  if (hasPermission === false) return <Text>No hay acceso a cámara</Text>;

  return (
    <View style={{ flex: 1, backgroundColor: "black" }}>
      <Camera
        ref={cameraRef}
        style={{ flex: 1 }}
        zoom={zoom}
      />

      {/* Logo Blink AI */}
      <Animated.Text
        style={[styles.logo, { opacity: blinkAnim }]}
      >
        Blink AI
      </Animated.Text>

      {/* Zoom */}
      <View style={styles.zoomContainer}>
        <Text style={{ color: "#fff" }}>{displayZoom}x</Text>
        <Slider
          style={{ width: width * 0.8 }}
          minimumValue={0}
          maximumValue={1}
          onValueChange={handleZoom}
        />
      </View>

      {/* Modos */}
      <View style={styles.modes}>
        <Text style={styles.mode}>VLOG</Text>
        <Text style={[styles.mode, { color: "yellow" }]}>AI CAM</Text>
        <Text style={styles.mode}>VIDEO</Text>
      </View>

      {/* Botón */}
      <View style={styles.controls}>
        <TouchableOpacity
          style={styles.shutter}
          onPress={takePhoto}
        />
        <TouchableOpacity onPress={() => navigation.navigate("More")}>
          <Text style={{ color: "#fff" }}>Más</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}

function MoreScreen() {
  return (
    <View style={styles.moreContainer}>
      {["Filtros de Cielo", "Pro", "4K Ultra Claridad", "Documentos"].map(
        (item, index) => (
          <View key={index} style={styles.iconBox}>
            <Text>{item}</Text>
          </View>
        )
      )}
    </View>
  );
}

export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator screenOptions={{ headerShown: false }}>
        <Stack.Screen name="Camera" component={CameraScreen} />
        <Stack.Screen name="More" component={MoreScreen} />
      </Stack.Navigator>
    </NavigationContainer>
  );
}

  logo: {
    position: "absolute",
    top: 50,
    left: 20,
    color: "white",
    fontSize: 18,
    fontWeight: "bold",
  },
  zoomContainer: {
    position: "absolute",
    bottom: 150,
    alignItems: "center",
    width: "100%",
  },
  controls: {
    position: "absolute",
    bottom: 50,
    width: "100%",
    alignItems: "center",
  },
  shutter: {
    width: 70,
    height: 70,
    borderRadius: 50,
    backgroundColor: "white",
    marginBottom: 10,
  },
  modes: {
    position: "absolute",
    bottom: 110,
    flexDirection: "row",
    justifyContent: "space-around",
    width: "100%",
  },
  mode: {
    color: "#aaa",
    fontSize: 16,
  },
  moreContainer: {
    flex: 1,

    flexWrap: "wrap",

  },
  iconBox: {

  },
